- Cosmos DB registration in ASP.NET Core (DI)

```csharp
// In Program.cs or Startup.cs
using Microsoft.Azure.Cosmos;

var builder = WebApplication.CreateBuilder(args);

var cosmosEndpoint = builder.Configuration["COSMOS_ENDPOINT"];
var cosmosKey = builder.Configuration["COSMOS_KEY"];

builder.Services.AddSingleton<CosmosClient>(_ =>
{
    var clientOptions = new CosmosClientOptions
    {
        ApplicationRegion = Regions.WestEurope // or a region of your choice
    };
    return new CosmosClient(cosmosEndpoint, cosmosKey, clientOptions);
});

var app = builder.Build();
// ... configure middleware, endpoints
```

- Simple repository pattern (pseudo)

```csharp
public interface ITaskRepository
{
    Task<TaskItem> CreateAsync(TaskItem item);
    Task<TaskItem> GetAsync(string id);
    Task<IEnumerable<TaskItem>> GetAllAsync();
    Task UpdateAsync(TaskItem item);
    Task DeleteAsync(string id);
}

public class CosmosTaskRepository : ITaskRepository
{
    private readonly CosmosClient _cosmos;
    private readonly Container _container;

    public CosmosTaskRepository(CosmosClient cosmos)
    {
        _cosmos = cosmos;
        _container = _cosmos.GetContainer("MyDb", "Tasks");
    }

    public async Task<TaskItem> CreateAsync(TaskItem item)
    {
        item.Id = Guid.NewGuid().ToString();
        var response = await _container.CreateItemAsync(item, new PartitionKey(item.PartitionKey));
        return response.Resource;
    }

    // Implement GetAsync, GetAllAsync, UpdateAsync, DeleteAsync similarly
}
```
